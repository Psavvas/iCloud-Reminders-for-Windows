# Phase 1 spike only. Not the app's dependency set.
#
# [cli] because the very next step in the README is `icloud auth login`, and the
# console script is a no-op without it: click, typer and rich are optional
# extras, so a plain install leaves `icloud` printing "CLI dependencies are not
# installed" and the keyring never gets seeded. The app itself never shells out
# to the CLI, which is why sidecar/pyproject.toml stays on the bare package.
pyicloud[cli]>=2.6.5

# keyring backend for Windows Credential Manager (pulled in by pyicloud, pinned
# here so the session-persistence check has somewhere to persist to)
keyring>=25.0

# zoneinfo has no tz database on Windows; without this, ZoneInfo("America/New_York")
# raises and the due-date test silently falls back to UTC, which would defeat
# the point of the tz-awareness check.
tzdata>=2024.1; sys_platform == "win32"
